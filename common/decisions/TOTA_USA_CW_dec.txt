OBM_unify = {
		icon = generic_unite_states

		visible = {
			OR = {
				tag = OBM
				original_tag = OBM
				has_country_leader = { name = "Barack H. Obama" ruling_only = yes }
			}
		}

		available = {
			has_war = no
			OR = {
				# Opción 1: Poseer todos los estados continentales americanos
				all_of_scopes = {
					array = global.USASTATES
					is_owned_by = ROOT
				}
				# Opción 2: Controlar Washington D.C. y haber derrotado a los rivales principales
				AND = {
					controls_state = 534 # Washington D.C.
					NOT = {
						country_exists = USA
						country_exists = CAL
						country_exists = ATI
						country_exists = NWF
						country_exists = USP
					}
				}
			}
		}

		cost = 50
		fire_only_once = yes

		ai_will_do = {
			base = 200
		}

		complete_effect = {
			# 1. Cambiar nombre y bandera a los Estados Unidos
			set_cosmetic_tag = USA

			# 2. Agregar reclamaciones y núcleos (cores) en todos los estados legítimos de EE. UU.
			every_state = {
				limit = {
					OR = {
						is_core_of = USA
						is_owned_by = ROOT
					}
				}
				add_core_of = ROOT
			}
			if = {
				limit = { has_idea = OBM_a_fragmented_movement }
				remove_ideas = OBM_a_fragmented_movement
			}
			if = {
				limit = { has_idea = OBM_the_hussein_hersey }
				remove_ideas = OBM_the_hussein_hersey
			}

			# 4. Bonificación nacional
			add_political_power = 150
			add_stability = 0.15
			add_war_support = 0.10
			every_country = {
				limit = { is_ai = no }
				news_event = {
					id = WNE.99
					days = 1
				}
			}
		}
	}
